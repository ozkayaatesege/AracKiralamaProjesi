package proje;

import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {

    private static CarInventory inventory = new CarInventory();
    private static Scanner scanner = new Scanner(System.in);
    private static final String Admin_Password="1905";

    public static void main(String[] args) {
        
        System.out.println("Sistem başlatılıyor, araç verileri yükleniyor...");
        loadCarsFromFile();

        System.out.println("=========================================");
        System.out.println("   ARAÇ KİRALAMA SİSTEMİNE HOŞ GELDİNİZ");
        System.out.println("=========================================");

        boolean exit = false;
        while (!exit) {
            System.out.println("\n--- ANA MENÜ ---");
            System.out.println("1 - Müsait Araçları Listele");
            System.out.println("2 - Araç Kirala");
            System.out.println("3 - Araç İade Et");
            System.out.println("4 - Yeni Araç Ekle");
            System.out.println("5 - Araç Sil");
            System.out.println("0 - Çıkış");
            System.out.print("Seçiminiz: ");

            int choice = -1;
            try {
                String input = scanner.nextLine();
                if(input.isEmpty()) continue;
                choice = Integer.parseInt(input);
            } catch (NumberFormatException e) {
                System.out.println("HATA: Lütfen sayısal bir değer giriniz!");
                continue;
            }

            switch (choice) {
                case 1:
                    inventory.listAvailableCars();
                    break;
                case 2:
                    rentCarProcess();
                    break;
                case 3:
                    returnCarProcess();
                    break;
                case 4:
                     if(adminDogrulaması()) {
                    	 addNewCarProcess();
                     }
                    break;
                case 5:
                    if(adminDogrulaması()) {
                    	deleteCarProcess();
                    }
                    break;
                case 0:
                    System.out.println("Çıkış yapılıyor. İyi günler!");
                    exit = true;
                    break;
                default:
                    System.out.println("Geçersiz seçim!");
            }
        }
    }

    private static void loadCarsFromFile() {
        File file = new File("cars.txt");

        try (Scanner fileScanner = new Scanner(file)) {
            while (fileScanner.hasNextLine()) {
                String line = fileScanner.nextLine();
                if (line.trim().isEmpty()) continue; 

                String[] data = line.split(",");
                if(data.length < 5) continue;

                String type = data[0].trim();
                String plaka = data[1].trim();
                String model = data[2].trim();
                double fiyat = Double.parseDouble(data[3].trim());

                if (type.equalsIgnoreCase("Gas")) {
                    double depo = Double.parseDouble(data[4].trim());
                    inventory.addCar(new GasCar(plaka, model, fiyat, depo));
                } else if (type.equalsIgnoreCase("Electric")) {
                    int batarya = Integer.parseInt(data[4].trim());
                    inventory.addCar(new ElectricCar(plaka, model, fiyat, batarya));
                }
            }
            System.out.println(">>> Veritabanı başarıyla yüklendi.");
        } catch (FileNotFoundException e) {
            System.out.println("UYARI: 'cars.txt' bulunamadı, sistem boş liste ile başlıyor.");
        } catch (Exception e) {
            System.out.println("HATA: Dosya okuma hatası: " + e.getMessage());
        }
    }
        private static boolean adminDogrulaması() {
        	System.out.println("\n--- GÜVENLİK KONTROLÜ ---");
            System.out.print("Admin şifresini giriniz: ");
            String password = scanner.nextLine();
            
            if(password.equals(Admin_Password)) {
            	System.out.println(">> Giriş Başarılı! Yönetici paneline yönlendiriliyorsunuz...");
            	return true;
            }else {
            	System.out.println("HATALI SİFRE!!!");
            	return false;
            }
        }
    


    private static void rentCarProcess() {
        System.out.println("\n--- ARAÇ KİRALAMA EKRANI ---");
        System.out.print("Plaka giriniz: ");
        String plaka = scanner.nextLine().trim().toUpperCase();

        Car car = inventory.getCarByPlate(plaka);

        if (car == null) {
            System.out.println("HATA: Araç bulunamadı!");
            return;
        }
        if (!car.isAvailable()) {
            System.out.println("HATA: Araç zaten kirada!");
            return;
        }


        System.out.println("Seçilen Araç: " + car.getModel());
        System.out.print("Müşteri Adı: ");
        String name = scanner.nextLine();
        System.out.print("Telefon: ");
        String phone = scanner.nextLine();
        
        int days = 0;
        try {
            System.out.print("Gün Sayısı: ");
            days = Integer.parseInt(scanner.nextLine());
        } catch (NumberFormatException e) {
            System.out.println("HATA: Gün sayısı rakam olmalı!");
            return;
        }

        Customer customer = new Customer(name, phone);
        

        Rental rental = new Rental(car, customer, days);
        

        System.out.println("\n--- ÖNİZLEME ---");
        System.out.println("Toplam Tutar: " + rental.getTotalPrice() + " TL");
        System.out.print("Onaylıyor musunuz? (E/H): ");
        String onay = scanner.nextLine();

        if (onay.equalsIgnoreCase("E")) {
            System.out.print("Kredi Kartı No: ");
            String cardNum = scanner.nextLine();
            
            boolean success = rental.executeRental(cardNum);
            
            if(success) {
                System.out.println(">>> Tebrikler, kiralama tamamlandı.");
            }
        } else {
            System.out.println("İşlem iptal edildi.");
        }
    }

    private static void returnCarProcess() {
        System.out.print("\nİade edilecek plaka: ");
        String plaka = scanner.nextLine().trim().toUpperCase();
        
        Car car = inventory.getCarByPlate(plaka);

        if (car != null) {
            car.returnCar(); 
        } else {
            System.out.println("HATA: Araç bulunamadı.");
        }
    }


    private static void addNewCarProcess() {
        System.out.println("\n--- YENİ ARAÇ EKLEME ---");
        try {
            System.out.print("Araç Tipi (Gas/Electric): ");
            String type = scanner.nextLine().trim();

            System.out.print("Plaka: ");
            String plaka = scanner.nextLine().trim().toUpperCase();

            System.out.print("Model: ");
            String model = scanner.nextLine();

            System.out.print("Günlük Fiyat: ");
            double fiyat = Double.parseDouble(scanner.nextLine());

            // Değişiklik Buradan Başlıyor: Nesneyi önce oluştur, sonra kaydet.
            Car newCar = null;

            if (type.equalsIgnoreCase("Gas")) {
                System.out.print("Depo Kapasitesi (Litre): ");
                double depo = Double.parseDouble(scanner.nextLine());
                newCar = new GasCar(plaka, model, fiyat, depo);
                
            } else if (type.equalsIgnoreCase("Electric")) {
                System.out.print("Batarya Kapasitesi (kWh): ");
                int batarya = Integer.parseInt(scanner.nextLine());
                newCar = new ElectricCar(plaka, model, fiyat, batarya);
            } else {
                System.out.println("Geçersiz araç tipi! (Gas veya Electric giriniz)");
                return; // Geçersizse işlem yapma
            }
            
            // Eğer araç başarıyla oluşturulduysa:
            if (newCar != null) {
                // 1. Listeye ekle (Program çalışırken görünsün)
                inventory.addCar(newCar);
                
                // 2. Dosyaya yaz (Kapatıp açınca silinmesin)
                saveCarToFile(newCar);
            }

        } catch (NumberFormatException e) {
            System.out.println("HATA: Sayısal değer girmeniz gereken yerde harf girdiniz.");
        } catch (Exception e) {
            System.out.println("HATA: Bir sorun oluştu: " + e.getMessage());
        }
    }


    private static void deleteCarProcess() {
        System.out.println("\n--- ARAÇ SİLME ---");
        System.out.print("Silinecek aracın plakası: ");
        String plaka = scanner.nextLine().trim().toUpperCase();
        
        // 1. Önce programın hafızasındaki listeden sil
        boolean isDeleted = inventory.deleteCar(plaka);
        
        // 2. Eğer listeden silme başarılıysa, dosyadan da sil
        if (isDeleted) {
            deleteCarFromFile(plaka);
        }
    }
    
    private static void saveCarToFile(Car car) {
    	try (FileWriter fw=new FileWriter("cars.txt",true)){
    		String line="";
    		
    		if(car instanceof GasCar) {
    			GasCar gc=(GasCar) car;
    			line="Gas,"+ gc.getPlakaKodu()+","+ gc.getModel()+","+gc.getGunlukFiyat()+","+gc.getDepoKapasitesi();
    		}else if(car instanceof ElectricCar) {
    			ElectricCar ec=(ElectricCar) car;
    			line="Electric,"+ ec.getPlakaKodu()+","+ec.getModel()+","+ec.getGunlukFiyat()+","+ec.getBataryaKapasitesi();
    		}
    		fw.write(line+"\n");
            System.out.println(">> [Sistem] Araç 'cars.txt' dosyasına kalıcı olarak kaydedildi.");
    	}
    	catch (IOException e) {
            System.out.println("HATA: Dosyaya yazarken sorun oluştu: " + e.getMessage());
        }
    }
    
    private static void deleteCarFromFile(String targetPlaka) {
        File file = new File("cars.txt");
        List<String> remainingLines = new ArrayList<>();

        // 1. ADIM: Dosyayı oku, silinecek araç HARİÇ diğerlerini listeye al
        try (Scanner fileScanner = new Scanner(file)) {
            while (fileScanner.hasNextLine()) {
                String line = fileScanner.nextLine();
                if (line.trim().isEmpty()) continue;

                // Satırı parçala ve plakayı kontrol et
                String[] data = line.split(",");
                if (data.length >= 2) {
                    String currentPlaka = data[1].trim();
                    
                    // Eğer bu satırdaki plaka, silmek istediğimiz plaka DEĞİLSE listeye ekle
                    if (!currentPlaka.equalsIgnoreCase(targetPlaka)) {
                        remainingLines.add(line);
                    }
                }
            }
        } catch (FileNotFoundException e) {
            System.out.println("HATA: Dosya bulunamadı.");
            return;
        }

        // 2. ADIM: Dosyayı SIFIRDAN (append: false) tekrar yaz
        try (FileWriter writer = new FileWriter("cars.txt", false)) { 
            for (String line : remainingLines) {
                writer.write(line + "\n"); // Her satırdan sonra alt satıra geç
            }
            System.out.println(">> [Sistem] Dosya güncellendi, araç kayıtlardan tamamen silindi.");
        } catch (IOException e) {
            System.out.println("HATA: Dosya güncellenirken hata oluştu: " + e.getMessage());
        }
    }
}
