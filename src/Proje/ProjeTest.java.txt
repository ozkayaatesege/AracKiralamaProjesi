package proje;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

class ProjeTest {

    @Test
    void testGasCarPrice() {

        Car gasCar = new GasCar("34 GAS 01", "Fiat Egea", 1000.0, 50.0);

        double beklenenFiyat = 5000.0;
        double gerceklesenFiyat = gasCar.calculatePrice(5);
        
        assertEquals(beklenenFiyat, gerceklesenFiyat, "Benzinli araç fiyatı hatalı hesaplandı!");
    }

    @Test
    void testElectricCarDiscount() {

        Car electricCar = new ElectricCar("34 EV 01", "Tesla", 1000.0, 75);
        
        double beklenenFiyat = 4000.0;
        double gerceklesenFiyat = electricCar.calculatePrice(5);
        
        assertEquals(beklenenFiyat, gerceklesenFiyat, "Elektrikli araç indirimi uygulanmadı!");
    }


    @Test
    void testRentAndReturnLogic() {
        Car car = new GasCar("06 ANK 06", "Ford", 500, 45);
        Customer customer = new Customer("Test Müşteri", "555-0000");


        assertTrue(car.isAvailable(), "Araç ilk oluşturulduğunda müsait olmalı.");


        car.rent(customer, 3);
        

        assertFalse(car.isAvailable(), "Kiralandıktan sonra araç müsait görünüyor (Hata)!");


        car.returnCar();
        

        assertTrue(car.isAvailable(), "İade edilen araç tekrar müsait olmalı.");
    }
    

    @Test
    void testPaymentProcess() {

        Payment payment = new Payment(1000.0);
        

        boolean sonuc = payment.processPayment("Ahmet Test", "1234123412341234");
        
        assertTrue(sonuc, "Geçerli kart ile ödeme başarısız oldu!");
    }
    

    @Test
    void testInventoryAdd() {
        CarInventory inv = new CarInventory();
        Car car = new GasCar("34 ABC 34", "Test Araba", 100, 50);
        

        inv.addCar(car);
        

        Car foundCar = inv.getCarByPlate("34 ABC 34");
        
        assertNotNull(foundCar, "Galeriye eklenen araç bulunamadı!");
        assertEquals("Test Araba", foundCar.getModel(), "Galeriden gelen araç modeli yanlış.");
    }
    
    @Test
    void testLongTermDiscount() {
        Car car = new GasCar("34 INDIRIM", "BMW", 1000.0, 60.0);
        Customer customer = new Customer("Test Müşteri", "555-1234");
        
        Rental rental = new Rental(car, customer, 10);
        
        double beklenenFiyat = 9000.0; 
        double gerceklesenFiyat = rental.getTotalPrice();
        
        assertEquals(beklenenFiyat, gerceklesenFiyat, "5 gün üzeri %10 indirim uygulanmadı!");
    }
}
