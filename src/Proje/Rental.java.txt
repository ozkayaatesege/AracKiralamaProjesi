package proje;

public class Rental {
    private Car car;
    private Customer customer;
    private int days;
    private double totalPrice;

    public Rental(Car car, Customer customer, int days) {
        this.car = car;
        this.customer = customer;
        this.days = days;
        this.totalPrice = car.calculatePrice(days);
    }

    public boolean executeRental(String creditCardNumber) {
        System.out.println(">>> Kiralama işlemi başlatılıyor...");

        Payment payment = new Payment(totalPrice);
        boolean paymentSuccess = payment.processPayment(customer.getName(), creditCardNumber);

        if (paymentSuccess) {
            car.rent(customer, days);
            return true;
        } else {
            System.out.println("HATA: Ödeme alınamadığı için kiralama yapılamadı.");
            return false;
        }
    }

    public double getTotalPrice() {
        return totalPrice;
    }
}
